\newcounter{questao}

\newenvironment{myenum}
{
    \begin{enumerate}[leftmargin = *, label=\textbf{\arabic*.}]
    \setcounter{enumi}{\thequestao}
    
}
{
    \end{enumerate}
    \setcounter{questao}{\theenumi}
}

\newenvironment{manualenum}[1][\theenumi]
{
    \begin{enumerate}[leftmargin = *, label=\textbf{\arabic*.}]
    \setcounter{enumi}{#1}
    \addtocounter{enumi}{-1}
    
}
{
    \end{enumerate}
}
